--[[
    INFANTRY TOOL v4 on weapon_bugbait - a soldier's build tool with its own spawn menu.

    Holding weapon_bugbait:
      Q          - open / close the tool menu (2 categories: TOOLS = mines, charges, guns; PROPS = fortifications)
      LMB        - place the selected item (ghost preview: green = ok, red = not allowed)
      RMB        - remove your object under the crosshair (or cancel an unfinished tripwire)
      R / Shift+R- rotate (+/-). Trenches: 90 deg on their grid
      G          - grid on/off for props.   Shift+G - reset the trench grid
      X          - detonate all C4

    TOOLS: AT mine, AP mine, Tripwire grenade (2 clicks), Claymore, C4 (X), Smoke pot,
           MANNED Machine gun, MANNED Sniper rifle.
    PROPS: Czech hedgehog, Dragon's teeth, Trench pieces on their own grid (straight / corner / T /
           cross / end / exit ramp), Sandbags (wall / low / arc / corner - round bags, STACKABLE),
           MG nest, Barbed wire, Watchtower (stairs + railing), Pillbox, Concrete T-wall,
           Log barricade, Supply crate (stackable).

    Stacking: aim at a sandbag piece / crate with another stackable item selected -> the ghost
    snaps exactly on top of it (same direction, R turns it). Build walls as high as you want.

    Manned guns (anyone can use them, not only you):
      walk to the gun and press E on the seat behind it (a little ammo box).
      LMB fire   RMB zoom (sniper: 2 zoom levels + scope)   R reload   E leave.
      The gun turns to where you look (limited arc + turn speed), MG overheats, ammo + reload.
      Bullets: game.bulletDamage from the gun muzzle. Server limits: 40 bullets/s and
      100 damage/s per chip owner (SF burst limits) - one MG at full fire uses ~90 dmg/s.

    How it works:
      * Everything is a custom prop (prop.createCustom): real collision, no hologram limit.
        Every client builds a textured mesh per item ONCE (in a coroutine, no CPU spikes)
        and applies it with Entity:setMesh. The gun body is 1 server hologram with a mesh.
      * All text is ASCII (no Cyrillic in the SF editor / Roboto font).
]]

local V = Vector

local function clamp(v, a, b) return v < a and a or (v > b and b or v) end

local function holdsTool(ply)
    if not isValid(ply) or not ply:isAlive() or ply:inVehicle() then return false end
    local w = ply:getActiveWeapon()
    return isValid(w) and w:getClass() == WEAPON_CLASS
end

-- ============================ GEOMETRY (shared) ============================
-- prims: {t="hex", p={8 corners: bottom 1-4, top 5-8}} | {t="cyl", c=bottomCenter, r, h, n}
--        {t="ocyl", a, b, r, n} (cylinder between two points) | {t="quads", q={{a,b,c,d}...}}
--        {t="sq", q={{a,b,c,d, na,nb,nc,nd}...}} (smooth quads with vertex normals, visual only)
local function hex(list, p) list[#list + 1] = { t = "hex", p = p } end

local function box(list, mn, mx)
    hex(list, {
        V(mn.x, mn.y, mn.z), V(mx.x, mn.y, mn.z), V(mx.x, mx.y, mn.z), V(mn.x, mx.y, mn.z),
        V(mn.x, mn.y, mx.z), V(mx.x, mn.y, mx.z), V(mx.x, mx.y, mx.z), V(mn.x, mx.y, mx.z),
    })
end

-- oriented box: center c, axes ax/ay/az (unit), half sizes hx/hy/hz
local function obox(list, c, ax, ay, az, hx, hy, hz)
    local function P(sx, sy, sz) return c + ax * (sx * hx) + ay * (sy * hy) + az * (sz * hz) end
    hex(list, { P(-1, -1, -1), P(1, -1, -1), P(1, 1, -1), P(-1, 1, -1), P(-1, -1, 1), P(1, -1, 1), P(1, 1, 1), P(-1, 1, 1) })
end

-- box rotated by yaw around Z (for arcs)
local function ybox(list, c, yaw, hx, hy, hz)
    local a = Angle(0, yaw, 0)
    obox(list, c, a:getForward(), -a:getRight(), V(0, 0, 1), hx, hy, hz)
end

-- beam box between two points (planks, braces)
local function beam(list, a, b, hw, hh)
    local d = b - a
    local len = d:getLength()
    local f = d / len
    local s = f:cross(math.abs(f.z) < 0.95 and V(0, 0, 1) or V(1, 0, 0)):getNormalized()
    local u = s:cross(f):getNormalized()
    obox(list, (a + b) / 2, f, s, u, len / 2, hw, hh)
end

-- frustum: base bx*by at z0, top tx*ty at z0+h
local function frustum(list, cx, cy, z0, bx, by, tx, ty, h)
    hex(list, {
        V(cx - bx, cy - by, z0), V(cx + bx, cy - by, z0), V(cx + bx, cy + by, z0), V(cx - bx, cy + by, z0),
        V(cx - tx, cy - ty, z0 + h), V(cx + tx, cy - ty, z0 + h), V(cx + tx, cy + ty, z0 + h), V(cx - tx, cy + ty, z0 + h),
    })
end

local function cyl(list, c, r, h, n) list[#list + 1] = { t = "cyl", c = c, r = r, h = h, n = n or 12 } end
local function ocyl(list, a, b, r, n) list[#list + 1] = { t = "ocyl", a = a, b = b, r = r, n = n or 10 } end

local function ocylBasis(a, b)
    local d = (b - a):getNormalized()
    local e1 = d:cross(math.abs(d.z) < 0.9 and V(0, 0, 1) or V(1, 0, 0)):getNormalized()
    local e2 = d:cross(e1):getNormalized()
    return e1, e2
end

-- ROUND sandbag: superellipsoid pillow with smooth normals (visual only)
local BAG_NU, BAG_NV = 10, 5
local function spow(v, e) if v < 0 then return -((-v) ^ e) end return v ^ e end
local function bag(list, c, yaw, l, w, h)
    local an = Angle(0, yaw, 0)
    local ax, ay, az = an:getForward(), -an:getRight(), V(0, 0, 1)
    local A, B, C = l * 0.5, w * 0.5, h * 0.5
    local E1, E2 = 0.55, 0.32        -- vertical / plan squareness (1 = ellipsoid, -> 0 = box)
    local P, N = {}, {}
    for j = 0, BAG_NV do
        local v = -math.pi / 2 + math.pi * j / BAG_NV
        local cv, sv = math.cos(v), math.sin(v)
        if j == 0 or j == BAG_NV then cv = 0 end
        P[j], N[j] = {}, {}
        for i = 0, BAG_NU - 1 do
            local u = math.pi * 2 * i / BAG_NU + math.pi / BAG_NU
            local cu, su = math.cos(u), math.sin(u)
            local x = A * spow(cv, E1) * spow(cu, E2)
            local y = B * spow(cv, E1) * spow(su, E2)
            local z = C * spow(sv, E1)
            P[j][i] = c + ax * x + ay * y + az * (z + C)
            local nx = spow(cv, 2 - E1) * spow(cu, 2 - E2) / A
            local ny = spow(cv, 2 - E1) * spow(su, 2 - E2) / B
            local nz = spow(sv, 2 - E1) / C
            local n = ax * nx + ay * ny + az * nz
            if n:getLength() < 1e-6 then n = az * (sv < 0 and -1 or 1) end
            N[j][i] = n:getNormalized()
        end
    end
    local q = {}
    for j = 0, BAG_NV - 1 do
        for i = 0, BAG_NU - 1 do
            local i2 = (i + 1) % BAG_NU
            q[#q + 1] = { P[j][i], P[j][i2], P[j + 1][i2], P[j + 1][i], N[j][i], N[j][i2], N[j + 1][i2], N[j + 1][i] }
        end
    end
    list[#list + 1] = { t = "sq", q = q }
end

-- row(s) of sandbags along X from x0 to x1 at y, layer count (bricked)
local BAG_L, BAG_W, BAG_H = 22, 13, 8
local function bagRow(list, x0, x1, y, z, layers)
    for k = 0, layers - 1 do
        local off = (k % 2 == 1) and BAG_L * 0.5 or 0
        local x = x0 + BAG_L * 0.5 + off
        while x + BAG_L * 0.5 <= x1 + 0.1 do
            bag(list, V(x, y, z + k * (BAG_H - 1)), 0, BAG_L, BAG_W, BAG_H)
            x = x + BAG_L
        end
    end
end
local function bagRowY(list, y0, y1, x, z, layers)
    for k = 0, layers - 1 do
        local off = (k % 2 == 1) and BAG_L * 0.5 or 0
        local y = y0 + BAG_L * 0.5 + off
        while y + BAG_L * 0.5 <= y1 + 0.1 do
            bag(list, V(x, y, z + k * (BAG_H - 1)), 90, BAG_L, BAG_W, BAG_H)
            y = y + BAG_L
        end
    end
end
local function stackH(layers) return layers * (BAG_H - 1) + 1 end

-- convex point list for a prim (server collision)
local function primPoints(pr)
    if pr.t == "hex" then return pr.p end
    if pr.t == "cyl" then
        local pts = {}
        for i = 0, pr.n - 1 do
            local a = i / pr.n * math.pi * 2
            local x, y = math.cos(a) * pr.r, math.sin(a) * pr.r
            pts[#pts + 1] = pr.c + V(x, y, 0)
            pts[#pts + 1] = pr.c + V(x, y, pr.h)
        end
        return pts
    end
    if pr.t == "ocyl" then
        local e1, e2 = ocylBasis(pr.a, pr.b)
        local pts = {}
        for i = 0, pr.n - 1 do
            local a = i / pr.n * math.pi * 2
            local o = e1 * (math.cos(a) * pr.r) + e2 * (math.sin(a) * pr.r)
            pts[#pts + 1] = pr.a + o
            pts[#pts + 1] = pr.b + o
        end
        return pts
    end
end

local function hedgehog(l, beamHalf, thick)
    local up = V(0, 0, 1)
    local s3, s23 = 1 / math.sqrt(3), math.sqrt(2 / 3)
    local zc = beamHalf * s3 + 1
    for k = 0, 2 do
        local a = math.rad(k * 120)
        local e = up * s3 + V(math.cos(a), math.sin(a), 0) * s23
        local p1 = e:cross(up):getNormalized()
        local p2 = e:cross(p1):getNormalized()
        obox(l, V(0, 0, zc), e, p1, p2, beamHalf, thick, thick)
    end
end

-- One cell = TRENCH_CELL. Channel along X (width TRENCH_WIDTH). Each closed side has a wall:
--   inner vertical face (wood posts) -> flat parapet with sandbags (TRENCH_DEPTH high)
--   -> outer earth slope down to the ground, TRENCH_SPILL beyond the cell (walkable)
-- camo net: sagging rope grid + leaf patches (visual only, bullets and players pass through)
local function camoNet(l, x0, x1, y0, y1, z, sag)
    local q = {}
    local nx, ny = 16, 8
    local function P(u, v) return V(x0 + (x1 - x0) * u, y0 + (y1 - y0) * v, z - sag * (1 - (2 * u - 1) ^ 2) * (1 - (2 * v - 1) ^ 2)) end
    local t = 0.5
    for j = 0, ny do
        for i = 0, nx - 1 do
            local a, b = P(i / nx, j / ny), P((i + 1) / nx, j / ny)
            q[#q + 1] = { a - V(0, t, 0), b - V(0, t, 0), b + V(0, t, 0), a + V(0, t, 0) }
        end
    end
    for i = 0, nx do
        for j = 0, ny - 1 do
            local a, b = P(i / nx, j / ny), P(i / nx, (j + 1) / ny)
            q[#q + 1] = { a - V(t, 0, 0), b - V(t, 0, 0), b + V(t, 0, 0), a + V(t, 0, 0) }
        end
    end
    local seed = 12345
    local function rnd() seed = (seed * 16807) % 2147483647 return seed / 2147483647 end
    for _ = 1, 90 do                                                   -- leaf patches
        local c = P(rnd(), rnd()) - V(0, 0, 0.4)
        local sz, a = 4 + rnd() * 6, rnd() * math.pi
        local d1 = V(math.cos(a), math.sin(a), 0) * sz
        local d2 = V(-math.sin(a), math.cos(a), 0) * (sz * 0.6)
        q[#q + 1] = { c - d1 - d2, c + d1 - d2 + V(0, 0, rnd() * 2 - 1), c + d1 + d2, c - d1 + d2 + V(0, 0, rnd() * 2 - 1) }
    end
    l[#l + 1] = { t = "quads", q = q }
end

-- cover: "open" | "net" (camo net on poles) | "roof" (plank roof, firing slit under it) | "closed" (roof, no slits)
local function trenchShape(kind, vis, cover)
    local l_
    local h   = TRENCH_CELL / 2
    local ch  = TRENCH_WIDTH / 2        -- channel half width
    local tw  = 12                      -- parapet top width
    local H   = TRENCH_DEPTH
    local out = h + TRENCH_SPILL        -- slope foot
    local Z0  = -8                      -- dig a bit into the ground
    local c   = ch + tw
    local RZ  = H + ROOF_Z
    local closed = cover == "closed"
    local WT  = closed and RZ or H      -- closed: the walls go up to the roof (no slit)
    local function M(axis, u, v, z) if axis == "x" then return V(u, v, z) else return V(v, u, z) end end
    local function hx(axis, a0, a1, b0, b1, z0, z1)
        hex(l_, { M(axis, a0, b0, z0), M(axis, a1, b0, z0), M(axis, a1, b1, z0), M(axis, a0, b1, z0),
                  M(axis, a0, b0, z1), M(axis, a1, b0, z1), M(axis, a1, b1, z1), M(axis, a0, b1, z1) })
    end
    local function wall(axis, a0, a1, s)
        local b0, b1 = math.min(s * ch, s * c), math.max(s * ch, s * c)
        hx(axis, a0, a1, b0, b1, Z0, WT)                                    -- core
        local i, o = s * c, s * out                                          -- outer slope
        hex(l_, { M(axis, a0, i, Z0), M(axis, a1, i, Z0), M(axis, a1, o, Z0), M(axis, a0, o, Z0),
                  M(axis, a0, i, H),  M(axis, a1, i, H),  M(axis, a1, o, 0),  M(axis, a0, o, 0) })
        if vis then
            for row = 0, closed and 3 or 0 do                              -- round sandbags on the parapet
                local u = a0 + 11 + ((row % 2 == 1) and 11 or 0)
                while u + 11 <= a1 + 0.1 do
                    bag(l_, M(axis, u, s * (ch + tw / 2), H - 1 + row * 7), axis == "x" and 0 or 90, 22, 12, 8)
                    u = u + 22
                end
            end
            local u = a0 + 12
            while u <= a1 - 8 do                                            -- wooden posts on the inner face
                local p0, p1 = math.min(s * ch, s * (ch - 2)), math.max(s * ch, s * (ch - 2))
                hx(axis, u - 2, u + 2, p0, p1, 0, WT)
                u = u + 46
            end
        end
    end
    local function pillar(x0, x1, y0, y1) box(l_, V(x0, y0, Z0), V(x1, y1, WT)) end
    local function filler(sx, sy)
        hex(l_, { V(sx * out, sy * out, Z0), V(sx * c, sy * out, Z0), V(sx * c, sy * c, Z0), V(sx * out, sy * c, Z0),
                  V(sx * out, sy * out, 0),  V(sx * c, sy * out, 0),  V(sx * c, sy * c, H),  V(sx * out, sy * c, 0) })
    end
    local walls = {}
    return function(l)
        l_ = l
        if kind == "straight" or kind == "exit" then
            wall("x", -h, h, 1) wall("x", -h, h, -1) walls = { py = true, ny = true }
            if kind == "straight" and TRENCH_FIRE_STEP then
                box(l, V(-h, -ch, Z0), V(h, -ch + 22, 18))
            end
            if kind == "exit" then
                hex(l, { V(-h, 4, Z0), V(h, 4, Z0), V(h, ch, Z0), V(-h, ch, Z0),
                         V(-h, 4, 0),  V(h, 4, H),  V(h, ch, H),  V(-h, ch, 0) })
            end
        elseif kind == "corner" then
            wall("x", -h, h, -1) wall("y", -h, h, -1) walls = { ny = true, nx = true }
            pillar(ch, h, ch, h)
            filler(-1, -1)
        elseif kind == "tee" then
            wall("x", -h, h, -1) walls = { ny = true }
            pillar(-h, -ch, ch, h) pillar(ch, h, ch, h)
        elseif kind == "cross" then
            pillar(-h, -ch, -h, -ch) pillar(ch, h, -h, -ch) pillar(-h, -ch, ch, h) pillar(ch, h, ch, h)
        elseif kind == "end" then
            wall("x", -h, h, 1) wall("x", -h, h, -1) wall("y", -h, h, -1) walls = { py = true, ny = true, nx = true }
            filler(-1, -1) filler(-1, 1)
        end
        local d = ch + 6                                                    -- 4 posts stand on walls / pillars
        if cover == "net" and vis then
            for sx = -1, 1, 2 do for sy = -1, 1, 2 do
                box(l, V(sx * d - 1.5, sy * d - 1.5, H), V(sx * d + 1.5, sy * d + 1.5, H + 46))
            end end
            camoNet(l, -h - 4, h + 4, -h - 4, h + 4, H + 46, 16)
        elseif cover == "roof" or closed then
            if not vis then box(l, V(-h, -h, RZ), V(h, h, RZ + 4)) return end
            local x = -h
            while x < h - 1 do                                              -- planks over the whole cell
                box(l, V(x + 0.6, -h, RZ), V(math.min(x + 15, h) - 0.6, h, RZ + 3))
                x = x + 15
            end
            for s = -1, 1, 2 do                                             -- beams
                box(l, V(-h, s * d - 3, RZ - 6), V(h, s * d + 3, RZ))
                box(l, V(s * d - 3, -h, RZ - 7), V(s * d + 3, h, RZ - 6))
            end
            for sx = -1, 1, 2 do for sy = -1, 1, 2 do
                box(l, V(sx * d - 2.5, sy * d - 2.5, H), V(sx * d + 2.5, sy * d + 2.5, RZ - 6))
            end end
            if walls.py then bagRow(l, -h, h, c - 4, RZ + 3, 1) end          -- bags on the roof over the walls
            if walls.ny then bagRow(l, -h, h, -c + 4, RZ + 3, 1) end
            if walls.nx then bagRowY(l, -h + 22, h - 22, -c + 4, RZ + 3, 1) end
        end
    end
end

-- 2x2 trench shelter: sits on the trench grid (4 cells), doors line up with trench channels
-- doors = list of sides with a door ("nx" / "px"), the door is in the cell row y = -CELL/2
local SHELTER_IN = TRENCH_CELL - 36
local function shelterShape(doors, vis, cover)
    return function(l)
        local B, H, tw, Z0 = TRENCH_CELL, TRENCH_DEPTH, 12, -8
        local S, ch = SHELTER_IN, TRENCH_WIDTH / 2
        local c = S + tw
        local out = B + TRENCH_SPILL
        local RZ = H + ROOF_Z
        local closed = cover == "closed"
        local WT = closed and RZ or H
        local DY = -B / 2
        local door = {}
        for _, k in ipairs(doors) do door[k] = true end
        local function M(axis, u, v, z) if axis == "x" then return V(u, v, z) else return V(v, u, z) end end
        local function seg(axis, s, u0, u1, f0, f1)                       -- wall core + outer slope piece
            local b0, b1 = math.min(s * S, s * c), math.max(s * S, s * c)
            hex(l, { M(axis, u0, b0, Z0), M(axis, u1, b0, Z0), M(axis, u1, b1, Z0), M(axis, u0, b1, Z0),
                     M(axis, u0, b0, WT), M(axis, u1, b0, WT), M(axis, u1, b1, WT), M(axis, u0, b1, WT) })
            local i, o = s * c, s * out
            hex(l, { M(axis, f0, i, Z0), M(axis, f1, i, Z0), M(axis, f1, o, Z0), M(axis, f0, o, Z0),
                     M(axis, u0, i, H),  M(axis, u1, i, H),  M(axis, f1, o, 0),  M(axis, f0, o, 0) })
            if vis then
                for row = 0, closed and 3 or 0 do
                    local u = u0 + 11 + ((row % 2 == 1) and 11 or 0)
                    while u + 11 <= u1 + 0.1 do
                        bag(l, M(axis, u, s * (S + tw / 2), H - 1 + row * 7), axis == "x" and 0 or 90, 22, 12, 8)
                        u = u + 22
                    end
                end
                local u = u0 + 14
                while u <= u1 - 10 do
                    local p0, p1 = math.min(s * S, s * (S - 2)), math.max(s * S, s * (S - 2))
                    hex(l, { M(axis, u - 2, p0, 0), M(axis, u + 2, p0, 0), M(axis, u + 2, p1, 0), M(axis, u - 2, p1, 0),
                             M(axis, u - 2, p0, WT), M(axis, u + 2, p0, WT), M(axis, u + 2, p1, WT), M(axis, u - 2, p1, WT) })
                    u = u + 45
                end
            end
        end
        local function side(axis, s, hasDoor)
            if not hasDoor then seg(axis, s, -c, c, -out, out) return end
            seg(axis, s, -c, DY - ch, -out, DY - ch)
            seg(axis, s, DY + ch, c, DY + ch, out)
        end
        side("x", 1, false) side("x", -1, false)                          -- +Y / -Y walls
        side("y", 1, door.px) side("y", -1, door.nx)                      -- +X / -X walls (doors)
        if door.px then box(l, V(-S, S - 22, Z0), V(S, S, 18))            -- fire step
        else box(l, V(S - 22, -S, Z0), V(S, S, 18)) end
        if cover == "net" and vis then
            local d = S + 6
            for sx = -1, 1, 2 do for sy = -1, 1, 2 do
                box(l, V(sx * d - 1.5, sy * d - 1.5, H), V(sx * d + 1.5, sy * d + 1.5, H + 50))
            end end
            box(l, V(-1.5, -1.5, 0), V(1.5, 1.5, H + 44))
            camoNet(l, -B - 4, B + 4, -B - 4, B + 4, H + 50, 10)
        elseif cover == "roof" or closed then
            if not vis then box(l, V(-B, -B, RZ), V(B, B, RZ + 4)) return end
            local x = -B
            while x < B - 1 do
                box(l, V(x + 0.6, -B, RZ), V(math.min(x + 16, B) - 0.6, B, RZ + 3))
                x = x + 16
            end
            local d = S + 6
            for _, y in ipairs({ -d, 0, d }) do box(l, V(-B, y - 3, RZ - 6), V(B, y + 3, RZ)) end
            for sx = -1, 1, 2 do
                box(l, V(sx * 78 - 3, -3, 0), V(sx * 78 + 3, 3, RZ - 6))   -- 2 posts in the middle
                for sy = -1, 1, 2 do
                    box(l, V(sx * d - 2.5, sy * d - 2.5, H), V(sx * d + 2.5, sy * d + 2.5, RZ - 6))
                end
            end
            bagRow(l, -B, B, c - 4, RZ + 3, 1) bagRow(l, -B, B, -c + 4, RZ + 3, 1)
        end
    end
end

local function wireCoil(l, len, r)
    local q = {}
    local turns, seg = math.floor(len / 7), 10
    local thick = 0.35
    local prev
    for i = 0, turns * seg do
        local t = i / (turns * seg)
        local a = i / seg * math.pi * 2
        local p = V(-len / 2 + len * t, math.cos(a) * r, r + math.sin(a) * r)
        if prev then
            local d = (p - prev):getNormalized()
            local side = d:cross(V(0, 0, 1)):getNormalized() * thick
            local upv = d:cross(side):getNormalized() * thick
            q[#q + 1] = { prev - side, p - side, p + side, prev + side }
            q[#q + 1] = { prev - upv, p - upv, p + upv, prev + upv }
        end
        prev = p
    end
    l[#l + 1] = { t = "quads", q = q }
end

-- ---------------- watchtower ----------------
-- 128x128 platform at 196, railing 42 high. Stairs = a steep wall of small blocks, every row sticks out
-- a bit (rise 16.3 < step size 18, so you just walk up). Step collision lives in 2 extra hidden
-- custom props (a prop can only have 10 convexes).
local TW_P, TW_T, TW_R = 64, 196, 42
local TW_STEPS, TW_RUN, TW_Y0, TW_Y1 = 12, 9, 16, 60
local function stepX(k) return -TW_P - TW_RUN * (TW_STEPS - k + 1) end     -- front edge of step k
local function towerSteps(k0, k1)
    return function(l)
        for k = k0, k1 do
            box(l, V(stepX(k), TW_Y0, TW_T * (k - 1) / TW_STEPS), V(-TW_P, TW_Y1, TW_T * k / TW_STEPS))
        end
    end
end
local function towerShape(vis)
    return function(l)
        local P, T, R = TW_P, TW_T, TW_R
        local lh = 52
        for sx = -1, 1, 2 do for sy = -1, 1, 2 do                                  -- 4 legs
            box(l, V(sx * lh - 4, sy * lh - 4, 0), V(sx * lh + 4, sy * lh + 4, T - 8))
        end end
        box(l, V(-P, -P, T - 8), V(P, P, T))                                      -- platform
        box(l, V(P - 3, -P, T), V(P, P, T + R))                                   -- railings
        box(l, V(-P, P - 3, T), V(P - 3, P, T + R))
        box(l, V(-P, -P, T), V(P - 3, -P + 3, T + R))
        box(l, V(-P, -P + 3, T), V(-P + 3, TW_Y0 - 4, T + R))                     -- (opening for the stairs)
        if not vis then return end
        -- block stairs: every row = bricks with gaps, sticking out 1.5 past the collision
        local BW = 11
        for k = 1, TW_STEPS do
            local z0, z1 = T * (k - 1) / TW_STEPS, T * k / TW_STEPS
            local x0 = stepX(k)

local function upFrame(n) return n:getAngle() + Angle(90, 0, 0) end

-- angle with UP = n and forward as close as possible to yaw
local function surfaceAngle(n, yaw)
    local f0 = Angle(0, yaw, 0):getForward()
    local f = f0 - n * f0:dot(n)
    if f:getLength() < 0.01 then f = n:cross(V(0, 1, 0)) end
    return f:getNormalized():getAngleEx(n)
end

local function aimTrace(ply, range, filter)
    local eye = ply:getShootPos()
    local f = { ply }
    if filter then for _, e in ipairs(filter) do f[#f + 1] = e end end
    return trace.line(eye, eye + ply:getAimVector() * range, f, MASK.SHOT), eye
end

local function badSurface(tr)
    if not tr.Hit or tr.HitSky then return true end
    local e = tr.Entity
    return isValid(e) and (e:isPlayer() or e:isNPC() or e:isVehicle())
end

-- mortar: fixed tube angle, solve the muzzle speed: D*tan(th) - g/2*t^2 = dz, D = v*cos(th)*t
local function mortarSolve(from, to, pitchDeg, g)
    local dx, dy = to.x - from.x, to.y - from.y
    local D = math.sqrt(dx * dx + dy * dy)
    if D < 1 then return nil end
    local th = math.rad(pitchDeg)
    local k = D * math.tan(th) - (to.z - from.z)
    if k <= 0 then return nil end
    local t = math.sqrt(2 * k / g)
    local c = math.cos(th)
    return V(dx / D * c, dy / D * c, math.sin(th)), D / (t * c), t, D
end

-- grenade model fix (w_ models have offset origin / arbitrary axes)
local function nadeFix(ent)
    local mn, mx = ent:obbMins(), ent:obbMaxs()
    local sz, c = mx - mn, (mn + mx) * 0.5
    local rot, half
    if sz.z >= sz.x and sz.z >= sz.y then rot, half = Angle(0, 0, 0), sz.z * 0.5
    elseif sz.x >= sz.y then rot, half = Angle(-90, 0, 0), sz.x * 0.5
    else rot, half = Angle(0, 0, 90), sz.y * 0.5 end
    return { rot = rot, c = c, half = half }
end
local function nadeTransform(fix, center, frame)
    local _, ang = localToWorld(V(), fix.rot, V(), frame)
    local off = localToWorld(fix.c, Angle(), V(), ang)
    return center - off, ang
end

local function wireOk(a, b, ignore)
    local d = b - a
    local len = d:getLength()
    if len < 24 or len > 600 then return false, len end
    local tr = trace.line(a, b - d / len * 3, ignore, MASK.SOLID)
    return not tr.Hit, len
end

    end

                return
            end
            local ok, x = pcall(prop.createCustom, p:getPos(), p:getAngles(), chunks[i], true)
            if ok and isValid(x) then
                pcall(x.setNoDraw, x, true)
                o.extras[#o.extras + 1] = x
                extraOf[x] = p
                sendReg(x:entIndex(), 255, nil, o.eidx)
                timer.simple(0.27, function() nextOne(i + 1, 0) end)
            elseif tries < 10 then
                timer.simple(0.5, function() nextOne(i, tries + 1) end)
            else
                sendState("Extra collision failed: " .. string.sub(tostring(x), 1, 50))
            end
        end
        timer.simple(0.27, function() nextOne(1, 0) end)
    end

    -- trench grid cells covered by a piece (1 or 2x2)
    local function trenchCells(pos, ang, size)
        if (size or 1) < 2 then return { pos } end
        local f, r, h = ang:getForward(), ang:getRight(), TRENCH_CELL / 2
        return { pos + f * h + r * h, pos + f * h - r * h, pos - f * h + r * h, pos - f * h - r * h }
    end
    local function trenchOverlaps(pos, ang, it)
        local mine, hit = trenchCells(pos, ang, it.size), {}
        for e, o in pairs(objs) do
            if isValid(e) and o.it.place == "trench" then
                local found = false
                for _, a in ipairs(trenchCells(e:getPos(), e:getAngles(), o.it.size)) do
                    for _, b in ipairs(mine) do
                        if a:getDistance(b) < 40 then found = true end
                    end
                end
                if found then hit[#hit + 1] = e end
            end
        end
        return hit
    end

    local function doPlace(it, pos, ang, normal, lvl, rep)
        local now = timer.curtime()
        if now - lastPlace < PLACE_COOLDOWN * 0.8 then return end
        if MAX_OBJECTS > 0 and countObjs() >= MAX_OBJECTS then sendState("Object limit: " .. MAX_OBJECTS) return end
        if not prop.canSpawn() then sendState("Prop limit, wait") return end
        if pos:getDistance(O:getShootPos()) > PLACE_RANGE + 200 then sendState("Too far") return end
        if it.place == "trench" then
            local ov = trenchOverlaps(pos, ang, it)
            -- replacing an existing trench piece / adding a cover: same footprint -> remove the old one first
            if rep and rep > 0 and #ov == 1 and ov[1]:entIndex() == rep and (objs[ov[1]].it.size or 1) == (it.size or 1) then
                unregister(ov[1])
                ov[1]:remove()
            elseif #ov > 0 then
                sendState("Cell is taken") return
            end
        end
        lastPlace = now

        if it.kind == "tripwire" then
            if pendingTrip then return end
            local ok, p = pcall(prop.create, pos + normal * 4, upFrame(normal), it.model, true)
            if not ok or not isValid(p) then sendState("Can't create grenade") return end
            local fix = nadeFix(p)
            local center = pos + normal * (fix.half + 0.3)
            local pp, pa = nadeTransform(fix, center, upFrame(normal))
            pcall(function() p:setPos(pp) p:setAngles(pa) end)
            local tr = trace.line(pos + normal * 2, pos - normal * 4, { p, O })
            if isValid(tr.Entity) and not tr.HitWorld then pcall(p.setParent, p, tr.Entity) end
            objs[p] = { it = it, n = normal, la = p:worldToLocal(center + normal * fix.half) }
            pendingTrip = { prop = p, la = objs[p].la, n = normal, born = now }
            snd(p, "weapons/slam/mine_mode.wav", 70, 110, 0.8)
            sendState()
            return
        end

        local p, err, rest
        if it.kind == "gun" then p, err, rest = placeGun(it, pos, ang, lvl) else p, err, rest = placeMesh(it, pos, ang) end
        if not p then sendState("Spawn failed: " .. string.sub(err or "?", 1, 60)) return end
        local trap = it.logic and BOOM[it.logic] ~= nil
        local o = { it = it, eidx = p:entIndex(), armAt = trap and (now + ARM_TIME) or nil,
                    mdef = (it.kind == "gun" and lvl > 0 and it.mountDefs) and it.mountDefs[lvl] or it.mdef }
        objs[p] = o
        list[#list + 1] = p
        sendReg(o.eidx, o.mdef)
        spawnExtras(p, o, rest)
        local s = guns[p]
        if s and s.hidx then sendReg(s.hidx, it.gdef) end
        if it.logic == "laser" then                                -- red beam along the mine's up axis
            local a, up = p:getPos() + p:getUp() * 5, p:getUp()
            local tr = trace.line(a, a + up * LASER_RANGE, { p, O }, MASK.SHOT)
            o.beamLen = math.max(LASER_RANGE * tr.Fraction, 12)
            if hologram.canSpawn() then
                local w = hologram.create(a + up * (o.beamLen / 2), upFrame(up), HM .. "cylinder.mdl")
                if w then
                    pcall(w.setSize, w, V(0.3, 0.3, o.beamLen))
                    pcall(w.setColor, w, Color(255, 30, 20, 170))
                    pcall(w.setMaterial, w, "models/debug/debugwhite")
                    pcall(w.setParent, w, p)
                    o.wire = w
                end
            end
        elseif it.logic == "timed" then
            o.fuseAt, o.nextBeep, o.armAt = now + TIMED_FUSE, now, nil
        end
        if it.logic == "alarm" or it.logic == "radar" or it.logic == "decoy" then
            o.armAt = now + 1
            snd(p, "buttons/blip1.wav", 70, 120, 0.8)
        elseif it.logic == "timed" then
            snd(p, "weapons/c4/c4_plant.wav", 75, 100, 1)
        elseif trap then
            snd(p, "weapons/slam/mine_mode.wav", 70, 100, 0.7)
            timer.simple(ARM_TIME, function() if isValid(p) and objs[p] then snd(p, "buttons/blip1.wav", 65, 140, 0.6) end end)
        else
            snd(p, "physics/concrete/concrete_impact_soft" .. math.random(1, 3) .. ".wav", 70, 100, 0.7)
        end
        sendState()
    end

    local function stretchWire(b, hitNormal)
        local pt = pendingTrip
        if not pt or not isValid(pt.prop) then pendingTrip = nil sendState() return end
        local p = pt.prop
        local a = p:localToWorld(pt.la)
        local ok, len = wireOk(a, b, { p, O })
        if not ok then sendState(len > 600 and "Too long" or (len < 24 and "Too short" or "Something is in the way")) return end
        local o = objs[p]
        o.b = b
        o.armAt = timer.curtime() + 1.5
        local dir = (b - a) / len
        if hologram.canSpawn() then
            local w = hologram.create((a + b) / 2, upFrame(dir), HM .. "cylinder.mdl")
            if w then w:setSize(V(0.35, 0.35, len)) w:setColor(Color(80, 80, 70)) w:setParent(p) o.wire = w end
        end
        if hologram.canSpawn() then
            local an = hologram.create(b + hitNormal * 1.5, upFrame(hitNormal), HM .. "cylinder.mdl")
            if an then an:setSize(V(0.8, 0.8, 3)) an:setColor(Color(120, 115, 100)) o.anchor = an end
        end
        list[#list + 1] = p
        pendingTrip = nil
        snd(p, "weapons/slam/mine_mode.wav", 70, 90, 0.8)
        sendState()
    end

    local function removeAimed()
        if pendingTrip then
            local p = pendingTrip.prop
            pendingTrip = nil
            if isValid(p) then removeObj(p) end
            sendState("Cancelled")
            return
        end
        local tr = aimTrace(O, PLACE_RANGE + 200)
        local e = tr.Entity
        if isValid(e) and seatBase[e] then e = seatBase[e] end
        if isValid(e) and extraOf[e] then e = extraOf[e] end
        if isValid(e) and objs[e] then
            if objs[e].fuseAt and objs[e].it.logic ~= "timed" then return end   -- timed charge can be defused
            snd(e, "buttons/lever7.wav", 70, 110, 0.8)
            removeObj(e)
            sendState("Removed")
        else
            sendState("Look at your object")
        end
    end

    local function clearAll()
        pendingTrip = nil
        for e in pairs(objs) do
            unregister(e)
            if isValid(e) then e:remove() end
        end
        objs, list = {}, {}
        sendState("All removed")
    end

    local VEH_PAT = { "vehicle", "glide", "lvs", "lfs", "simfphys", "tank", "wac", "apc" }
    local function isVehicleLike(e)
        if seatBase[e] or extraOf[e] then return false end         -- our own gun seats / stairs
        if e:isPlayer() then
            local v = e:getVehicle()
            if isValid(v) and seatBase[v] then return false end
            return e:inVehicle()
        end
        if e:isVehicle() then return true end
        local c = e:getClass()
        for _, k in ipairs(VEH_PAT) do if string.find(c, k, 1, true) then return true end end
        if c == "prop_physics" or c == "starfall_prop" then
            if objs[e] then return false end
            local ok, m = pcall(function() return e:getPhysicsObject():getMass() end)
            return ok and m and m >= 500 and e:getVelocity():getLength() > 20
        end
        return false
    end
    local function isInfantry(e)
        if e:isPlayer() then return e:isAlive() and not e:inVehicle() and not (OWNER_SAFE and e == O) end
        return e:isNPC()
    end
    local function visible(a, e, self)
        local tr = trace.line(a, e:obbCenterW(), { self }, MASK.SHOT)
        return not tr.Hit or tr.Entity == e
    end

    local radarPrev, radarTick = 0, 0
    local function sendRadar(now)
        local pts, seen = {}, {}
        local function add(t)
            if not seen[t] and #pts < 24 then seen[t] = true pts[#pts + 1] = t:getPos() end
        end
        for e, o in pairs(objs) do
            if isValid(e) and o.armAt and now >= o.armAt then
                if o.it.logic == "radar" then
                    for _, t in ipairs(find.inSphere(e:getPos(), RADAR_RANGE)) do
                        if isValid(t) and t ~= O and ((t:isPlayer() and t:isAlive()) or t:isNPC()) then add(t) end
                    end
                elseif o.it.logic == "alarm" and o.alarmT and isValid(o.alarmT) and now < (o.alarmUntil or 0) then
                    add(o.alarmT)
                end
            end
        end
        if #pts == 0 and radarPrev == 0 then return end
        radarPrev = #pts
        net.start("it_radar")
        net.writeUInt(#pts, 6)
        for _, v in ipairs(pts) do net.writeVector(v) end
        net.send(O)
    end

    -- ---------------- manned guns: aim, fire, reload, heat ----------------
    local function approach(cur, target, step)
        if math.abs(target - cur) <= step then return target end
        return cur + (target > cur and step or -step)
    end

    local function afterShot(h, g, s, now)
        if g.bolt then
            timer.simple(0.45, function() if isValid(h) then snd(h, g.bolt, 70, 100, 0.9) end end)
        end
        if s.ammo <= 0 then
            s.reloadAt = now + g.reload
            snd(h, g.rel1, 70, 100, 1)
        end
    end

    local function gunShoot(base, s, now)
        local g, h = s.g, s.holo
        if game.canFireBullets and not game.canFireBullets(g.dmg, 1) then return end   -- SF bullet burst budget
        local ml = s.geo.muzzle
        if g.twin then
            s.side = not s.side
            if s.side then ml = s.geo.muzzle2 end
        end
        local muzzle = h:localToWorld(ml)
        local dir = h:getForward()
        local hitPos
        local ok = pcall(fireBullets, muzzle, dir, g.dmg, 1, g.force, 30000, V(g.spread, g.spread, 0), 0, base,
            function(att, tr) hitPos = tr.HitPos end)
        if not ok then s.nextFire = now + 0.25 return end
        s.ammo = s.ammo - 1
        s.heat = s.heat + g.heatPer
        s.nextFire = now + g.rate
        s.shots = s.shots + 1
        snd(h, g.snd, g.lvl, g.pitch + math.random(-4, 4), 1)
        pcall(effect.create, "MuzzleFlash", { origin = muzzle, angles = h:getAngles(), flags = g.flash }, true)
        if s.shots % g.tracerEvery == 0 then
            pcall(effect.create, g.tracer, { start = muzzle, origin = hitPos or (muzzle + dir * 8000), scale = 9000 }, true)
        end
        if not g.blast then
            pcall(effect.create, "RifleShellEject", { origin = h:localToWorld(V(-4, 3, 2)), angles = h:getAngles() + Angle(0, 90, 0) }, true)
        end
        -- explosive rounds (AT gun / flak): blast where the round hit (not right at the muzzle)
        if g.blast and hitPos and hitPos:getDistance(muzzle) > 120 then
            if g.blast.small then
                pcall(game.blastDamage, hitPos, g.blast.r, g.blast.d)
                pcall(effect.create, "Explosion", { origin = hitPos, magnitude = 1, scale = 0.5, radius = g.blast.r, flags = 4 }, true)
            else
                explodeAt(hitPos, g.blast.r, g.blast.d, false, true)
            end
        end
        afterShot(h, g, s, now)
    end

    local function aimAt(base, s, worldPos, dt)
        local g = s.g
        local d = base:worldToLocal(worldPos) - s.pivot
        local ty = clamp(math.deg(math.atan2(d.y, d.x)), -g.yaw, g.yaw)
        local tp = clamp(-math.deg(math.atan2(d.z, math.sqrt(d.x * d.x + d.y * d.y))), -g.up, g.down)
        return ty, tp
    end
    local function turnTo(s, ty, tp, dt)
        local g, h = s.g, s.holo
        local ny, np = approach(s.yaw, ty, g.turn * dt), approach(s.pitch, tp, g.turn * dt)
        if math.abs(ny - s.yaw) > 0.01 or math.abs(np - s.pitch) > 0.01 then
            s.yaw, s.pitch = ny, np
            h:setLocalAngles(Angle(s.pitch, s.yaw, 0))
        end
    end

    -- mortar: fixed tube angle (gunner picks it), muzzle speed solved for the target point (from the drone camera)
    local function mortarFire(base, s, target, pitch, now)
        local g, h = s.g, s.holo
        if not isValid(h) or s.reloadAt or s.ammo <= 0 or now < s.nextFire then return end
        pitch = clamp(pitch, 20, 85)
        local piv = base:localToWorld(s.pivot)
        local D = V(target.x - piv.x, target.y - piv.y, 0):getLength()
        if D < MORTAR_MIN or D > MORTAR_MAX + 60 then return end
        local dir = mortarSolve(piv, target, pitch, g.mortar.g)
        if not dir then return end
        local muzzle = piv + dir * s.geo.muzzle.x
        local dir2, v0 = mortarSolve(muzzle, target, pitch, g.mortar.g)
        if not dir2 then return end
        local ld = base:worldToLocal(piv + dir2) - base:worldToLocal(piv)
        s.yaw, s.pitch = math.deg(math.atan2(ld.y, ld.x)), -pitch
        pcall(h.setLocalAngles, h, Angle(s.pitch, s.yaw, 0))
        spawnShell(muzzle, dir2 * v0, g.mortar, base, s.op)
        s.ammo = s.ammo - 1
        s.nextFire = now + g.rate
        snd(h, g.snd, g.lvl, g.pitch + math.random(-4, 4), 1)
        pcall(effect.create, "MuzzleFlash", { origin = muzzle, angles = dir2:getAngle(), flags = g.flash }, true)
        pcall(effect.create, "ThumperDust", { origin = base:getPos(), scale = 120 }, true)
        afterShot(h, g, s, now)
    end

    local function gunTick(base, s, now, dt)
        local g, h = s.g, s.holo
        local op = isValid(s.seat) and s.seat:getDriver() or nil
        if not isValid(op) then op = nil end
        setOperator(s, op)

        -- cooling / reload progress run even without a gunner
        s.heat = math.max(0, s.heat - g.cool * dt)
        if s.over and s.heat <= 0.35 then s.over = false end
        if s.reloadAt and now >= s.reloadAt then
            s.reloadAt = nil
            s.ammo = g.mag
            snd(h, g.rel2, 70, 100, 1)
        end
        if not op or not isValid(h) then return end

        if g.mortar then                                                -- tube follows the drone crosshair
            if s.mtarget then
                local d = base:worldToLocal(s.mtarget) - s.pivot
                turnTo(s, math.deg(math.atan2(d.y, d.x)), -(s.mpitch or 45), dt)
            end
            s.aPrev, s.rPrev = true, true
        else

        -- aim at what the gunner looks at (limited arc + turn speed)
        local eye, aim = op:getShootPos(), op:getAimVector()
        local tr = trace.line(eye, eye + aim * 30000, { op, base, s.seat }, MASK.SHOT)
        local d = base:worldToLocal(tr.HitPos) - s.pivot
        if d:getLength() < 60 then d = base:worldToLocal(base:getPos() + aim) end
        local ty = clamp(math.deg(math.atan2(d.y, d.x)), -g.yaw, g.yaw)
        local tp = clamp(-math.deg(math.atan2(d.z, math.sqrt(d.x * d.x + d.y * d.y))), -g.up, g.down)

            net.writeUInt(clamp(s.ammo, 0, 255), 8)
            net.writeUInt(clamp(math.floor(s.heat * 255), 0, 255), 8)

local UV = 1 / 48
-- one triangle (both windings). na/nb/nc = optional smooth vertex normals
-- GMod front face = clockwise seen from outside (SF's obj loader reverses the CCW obj order).
-- Solid parts (known outside via center / vertex normals) get ONE side -> half the triangles;
-- thin parts without an inside (nets, flag cloth) stay two-sided.
local function emitTri(out, a, b, c, center, na, nb, nc)
    local raw = (b - a):cross(c - a)
    if raw:getLength() < 1e-6 then return end
    local n = raw:getNormalized()
    if center and n:dot(a - center) < 0 then n = -n end
    if na and n:dot(na + nb + nc) < 0 then n = -n end
    local ax, ay, az = math.abs(n.x), math.abs(n.y), math.abs(n.z)
    local function vert(p, vn)
        local u, v
        if az >= ax and az >= ay then u, v = p.x * UV, p.y * UV
        elseif ax >= ay then u, v = p.y * UV, -p.z * UV
        else u, v = p.x * UV, -p.z * UV end
        return { pos = p, normal = vn or n, u = u, v = v }
    end
    local oriented = center ~= nil or na ~= nil
    if MESH_TWO_SIDED or not oriented or n:dot(raw) < 0 then
        out[#out + 1] = vert(a, na) out[#out + 1] = vert(b, nb) out[#out + 1] = vert(c, nc)
    end
    if MESH_TWO_SIDED or not oriented or n:dot(raw) > 0 then
        out[#out + 1] = vert(a, na) out[#out + 1] = vert(c, nc) out[#out + 1] = vert(b, nb)
    end
end
local function emitQuad(out, a, b, c, d, center)
    emitTri(out, a, b, c, center)
    emitTri(out, a, c, d, center)
end

local HEX_FACES = { { 1, 4, 3, 2 }, { 5, 6, 7, 8 }, { 1, 2, 6, 5 }, { 2, 3, 7, 6 }, { 3, 4, 8, 7 }, { 4, 1, 5, 8 } }
local function primTris(out, pr, mins, maxs)
    local function grow(p)
        mins.x, mins.y, mins.z = math.min(mins.x, p.x), math.min(mins.y, p.y), math.min(mins.z, p.z)
        maxs.x, maxs.y, maxs.z = math.max(maxs.x, p.x), math.max(maxs.y, p.y), math.max(maxs.z, p.z)
    end
    if pr.t == "hex" then
        local c = V()
        for _, p in ipairs(pr.p) do c = c + p grow(p) end
        c = c / 8
        for _, f in ipairs(HEX_FACES) do emitQuad(out, pr.p[f[1]], pr.p[f[2]], pr.p[f[3]], pr.p[f[4]], c) end
    elseif pr.t == "cyl" then
        local c = pr.c + V(0, 0, pr.h / 2)
        local top, bot = pr.c + V(0, 0, pr.h), pr.c
        grow(pr.c - V(pr.r, pr.r, 0)) grow(pr.c + V(pr.r, pr.r, pr.h))
        for i = 0, pr.n - 1 do
            local a1, a2 = i / pr.n * math.pi * 2, (i + 1) / pr.n * math.pi * 2
            local o1, o2 = V(math.cos(a1) * pr.r, math.sin(a1) * pr.r, 0), V(math.cos(a2) * pr.r, math.sin(a2) * pr.r, 0)
            emitQuad(out, bot + o1, bot + o2, top + o2, top + o1, c)
            emitTri(out, top, top + o1, top + o2, c)
            emitTri(out, bot, bot + o2, bot + o1, c)
        end
    elseif pr.t == "ocyl" then
        local e1, e2 = ocylBasis(pr.a, pr.b)
        local c = (pr.a + pr.b) / 2
        for i = 0, pr.n - 1 do
            local a1, a2 = i / pr.n * math.pi * 2, (i + 1) / pr.n * math.pi * 2
            local o1 = e1 * (math.cos(a1) * pr.r) + e2 * (math.sin(a1) * pr.r)
            local o2 = e1 * (math.cos(a2) * pr.r) + e2 * (math.sin(a2) * pr.r)
            emitQuad(out, pr.a + o1, pr.a + o2, pr.b + o2, pr.b + o1, c)
            emitTri(out, pr.b, pr.b + o1, pr.b + o2, c)
            emitTri(out, pr.a, pr.a + o2, pr.a + o1, c)
            grow(pr.a + o1) grow(pr.b + o1)
        end
    elseif pr.t == "quads" then
        for _, q in ipairs(pr.q) do
            emitQuad(out, q[1], q[2], q[3], q[4], nil)
            grow(q[1]) grow(q[3])
        end
    elseif pr.t == "sq" then
        for _, q in ipairs(pr.q) do
            emitTri(out, q[1], q[2], q[3], nil, q[5], q[6], q[7])
            emitTri(out, q[1], q[3], q[4], nil, q[5], q[7], q[8])
            grow(q[1]) grow(q[3])
        end
    end
end

-- meshes are built in a coroutine: big meshes (nest, trenches) never hit the CPU quota
local function cpuBusy() return cpuUsed() > cpuMax() * (MESH_CPU + 0.1) end

local function buildMesh(d)
    local verts, mins, maxs
    -- one mesh can hold max 65535 vertices: too big -> rebuild with rounder-but-cheaper sandbags
    for _, q in ipairs({ { 10, 5 }, { 8, 4 }, { 6, 3 } }) do
        BAG_NU, BAG_NV = q[1], q[2]
        local prims = {}
        d.fn(prims)
        verts = {}
        mins, maxs = V(1e9, 1e9, 1e9), V(-1e9, -1e9, -1e9)
        for _, pr in ipairs(prims) do
            primTris(verts, pr, mins, maxs)
            if cpuBusy() then coroutine.yield() end
        end
        if #verts <= 65535 then break end
    end
    BAG_NU, BAG_NV = 10, 5
    if #verts > 65535 then
        meshes[d.idx] = false
        print("[Infantry Tool] mesh too big for " .. d.key .. ": " .. #verts .. " vertices")
        return
    end
    local ok, m = pcall(mesh.createFromTable, verts, true)
    if ok and m then
        meshes[d.idx] = m
        bounds[d.idx] = { mins - V(2, 2, 2), maxs + V(2, 2, 2) }
    else
        meshes[d.idx] = false
        print("[Infantry Tool] mesh failed for " .. d.key .. ": " .. tostring(type(m) == "table" and (m.message or m.msg) or m))
    end
end
builder = coroutine.create(function()
    for _, d in ipairs(MDEFS) do
        buildMesh(d)
        coroutine.yield()
    end
end)
end

-- registry of placed objects: entIndex -> {d = mesh def, it = item or nil, applied}
local reg = {}
local alias = {}            -- hidden extra collision prop -> main object index
net.receive("it_reg", function()
    local idx, di = net.readUInt(16), net.readUInt(8)
    if di == 255 then alias[idx] = net.readUInt(16) return end
    local d = MDEFS[di]
    if d then reg[idx] = { d = d, it = d.it, applied = nil } end
end)
net.receive("it_unreg", function()
    local idx = net.readUInt(16)
    reg[idx] = nil
    for k, m in pairs(alias) do if m == idx then alias[k] = nil end end
end)
-- your object behind an entity (extra collision props map to their main object) -> entity, item
local function regOf(e)
    if not isValid(e) then return nil end
    local idx = e:entIndex()
    if alias[idx] then
        idx = alias[idx]
        e = entity(idx)
        if not isValid(e) then return nil end
    end
    local r = reg[idx]
    if not r or not r.it then return nil end
    return e, r.it
end

do
local function applyMesh(e, d)
    local m = meshes[d.idx]
    if not m then return false end
    local ok = pcall(function()
        e:setMesh(m)
        local mat = getMat(d.mat)
        if mat then e:setMeshMaterial(mat) end
        local b = bounds[d.idx]
        e:setRenderBounds(b[1], b[2])
    end)
    return ok
end

local nextApply = 0
hook.add("think", "it_meshes", function()
    if coroutine.status(builder) ~= "dead" then
        -- mesh.createFromTable(threaded) yields every 100 vertices -> resume MANY times per frame
        -- until this frame's CPU budget is used (was 1 resume per frame = very slow).
        -- NOTE: SF coroutine.resume is NOT protected and returns the yielded values (no ok flag)
        local budget = cpuMax() * MESH_CPU
        local n = 0
        while coroutine.status(builder) ~= "dead" and cpuUsed() < budget and n < 2000 do
            n = n + 1
            local ok, err = pcall(coroutine.resume, builder)
            if not ok then print("[Infantry Tool] mesh builder error: " .. tostring(err)) break end
        end
        return
    end
    local now = timer.realtime()
    if now < nextApply then return end
    nextApply = now + 0.5
    for idx, r in pairs(reg) do
        local e = entity(idx)
        if isValid(e) and r.applied ~= e then
            local c = e:getClass()
            if (c == "starfall_prop" or c == "starfall_hologram") and applyMesh(e, r.d) then r.applied = e end
        end
    end
end)

end

-- trench grid cells covered by a piece (1 cell or 2x2 shelter, centre on a grid corner)
local function cellsOf(pos, ang, size)
    if (size or 1) < 2 then return { pos } end
    local f, r, h = ang:getForward(), ang:getRight(), TRENCH_CELL / 2
    return { pos + f * h + r * h, pos + f * h - r * h, pos - f * h + r * h, pos - f * h - r * h }
end
-- your trench pieces that overlap these cells -> list of { e, it }
local function trenchOverlaps(pos, ang, size)
    local mine, hit = cellsOf(pos, ang, size), {}
    for idx, r in pairs(reg) do
        if r.it and r.it.place == "trench" then
            local e = entity(idx)
            if isValid(e) then
                local found = false
                for _, a in ipairs(cellsOf(e:getPos(), e:getAngles(), r.it.size)) do
                    for _, b in ipairs(mine) do
                        if (a.x - b.x) ^ 2 + (a.y - b.y) ^ 2 < 1600 and math.abs(a.z - b.z) < 100 then found = true end
                    end
                end
                if found then hit[#hit + 1] = { e = e, it = r.it } end
            end
        end
    end
    return hit
end

-- prop grid frame: aligned to your nearest grid object (so walls line up), else the world grid
local function propFrame(hit)
    local best, bd = nil, 300
    for idx, r in pairs(reg) do
        if r.it and r.it.grid then
            local e = entity(idx)
            if isValid(e) then
                local ep = e:getPos()
                local d = math.sqrt((ep.x - hit.x) ^ 2 + (ep.y - hit.y) ^ 2)
                if d < bd and math.abs(ep.z - hit.z) < 120 then best, bd = e, d end
            end
        end
    end
    if best then return best:getPos(), best:getAngles().y, true end
    return V(0, 0, 0), 0, false
end

-- gun aimed at one of your trenches / fortifications -> mount plan: fills p.pos / p.ang / p.lvl, returns true
--   trench / shelter: post next to the wall you look towards, barrel just over the parapet
--   walls (bags, hesco, logs, crates, T-wall, nest, pillbox...): behind the wall on your side, post height = wall + 12
--   watchtower: on the deck, post over the railing
local function mountPlan(it, tr, eyeYaw, p)
    local te, tit = regOf(tr.Entity)
    if not te then return false end
    local mortar = it.gun == "mortar"
    local ta = te:getAngles()
    if tit.place == "trench" then
        if mortar then return false end
        if not tit.walls then
            if tit.cover == "closed" then
                p.ok, p.reason = false, "Closed - no firing slits"
                p.pos, p.ang = tr.HitPos, Angle(0, eyeYaw, 0)
                return true
            end
            return false
        end
        local k = math.round(((eyeYaw - ta.y) % 360) / 90) % 4          -- 0 +X, 1 +Y, 2 -X, 3 -Y
        local d = Angle(0, k * 90, 0):getForward()
        local side = V(-d.y, d.x, 0)
        local lp = te:worldToLocal(tr.HitPos)
        local along = clamp(lp:dot(side), -(tit.span - 30), tit.span - 30)
        local lpos = d * (tit.inner - MOUNT_FRONT) + side * along
        p.pos = te:localToWorld(V(lpos.x, lpos.y, 0))
        p.ang = Angle(0, ta.y + k * 90, 0)
        p.normal = te:getUp()
        p.lvl, p.mountName = MOUNT_TRENCH, "TRENCH MOUNT - faces the wall you look at"
        if not tit.walls[({ "px", "py", "nx", "ny" })[k + 1]] then p.ok, p.reason = false, "No trench wall that way" end
        return true
    end
    local m = tit.mnt
    if not m then return false end
    if m.deck then
        local lp = te:worldToLocal(tr.HitPos)
        if lp.z < m.deck - 10 then return false end
        p.pos = te:localToWorld(V(clamp(lp.x, -m.half, m.half), clamp(lp.y, -m.half, m.half), m.deck))
        p.ang = Angle(0, eyeYaw, 0)
        p.normal = te:getUp()
        p.lvl = mortar and 0 or mountLevelFor(m.P)
        p.mountName = "TOWER DECK"
        return true
    end
    local lp, le = te:worldToLocal(tr.HitPos), te:worldToLocal(me:getShootPos())
    local best, bs, ba
    for _, f in ipairs(m.faces) do
        local ax = V(-f.d.y, f.d.x, 0)
        local along = clamp((lp.x - f.o.x) * ax.x + (lp.y - f.o.y) * ax.y, -f.span, f.span)
        local qx, qy = f.o.x + ax.x * along, f.o.y + ax.y * along
        local score = math.sqrt((lp.x - qx) ^ 2 + (lp.y - qy) ^ 2)
        if (le.x - f.o.x) * f.d.x + (le.y - f.o.y) * f.d.y <= 0 then score = score + 1000 end   -- you are not on that side
        if not best or score < bs then best, bs, ba = f, score, along end
    end
    if not best then return false end
    local ax = V(-best.d.y, best.d.x, 0)
    local P0 = m.P or (m.top + 12)
    local function at(back)
        local lpos = best.o + best.d * (best.half + back) + ax * ba
        local wp = te:localToWorld(V(lpos.x, lpos.y, 0))
        local topZ = te:localToWorld(V(lpos.x, lpos.y, P0)).z
        local g = trace.line(V(wp.x, wp.y, topZ), V(wp.x, wp.y, topZ - 400), { te, me }, MASK.SOLID)
        local gz = g.Hit and g.HitPos.z or wp.z
        return V(wp.x, wp.y, gz), topZ - gz
    end
    local pos, P = at(14)
    local lvl = mortar and 0 or mountLevelFor(P)
    if lvl == 0 then pos = at(30) end
    p.pos, p.lvl = pos, lvl
    p.ang = Angle(0, ta.y + math.deg(math.atan2(-best.d.y, -best.d.x)), 0)
    p.normal = V(0, 0, 1)
    p.mountName = lvl > 0 and ("WALL MOUNT - post " .. MOUNT_LEVELS[lvl]) or "BEHIND COVER"
    if not mortar and P > MOUNT_LEVELS[#MOUNT_LEVELS] + 14 then p.ok, p.reason = false, "Too high for a gun" end
    return true
end

local function computePlan(it)
    local eyeYaw = me:getEyeAngles().y
    if it.kind == "tripwire" and isValid(pendingProp) and pendingStart then
        local tr = aimTrace(me, PLACE_RANGE + 600, { pendingProp })
        local ok, len = wireOk(pendingStart, tr.HitPos, { pendingProp, me })
        ok = ok and not badSurface(tr)
        return { wire = true, ok = ok, a = pendingStart, b = tr.HitPos, normal = tr.HitNormal, len = len,
                 reason = ok and nil or (len > 600 and "Too long" or "Can't hook the wire here") }
    end
    local tr = aimTrace(me, PLACE_RANGE)
    if not tr.Hit or tr.HitSky then return { ok = false, reason = "Aim at a surface" } end
    if it.kind == "cover" then                       -- swap a placed trench piece for its covered version
        local te, tit = regOf(tr.Entity)
        if not te or tit.place ~= "trench" then return { ok = false, reason = "Aim at your trench piece" } end
        local v = TRENCH_VAR[tit.shape] and TRENCH_VAR[tit.shape][it.cover]
        if not v then return { ok = false, reason = "This piece can't get a cover" } end
        return { ok = v ~= tit, reason = (v == tit) and "Already like this" or nil, pos = te:getPos(), ang = te:getAngles(),
                 normal = te:getUp(), replace = te:entIndex(), show = v }
    end
    local n = tr.HitNormal
    local p = { ok = true, normal = n, pos = tr.HitPos }
    if badSurface(tr) then p.ok, p.reason = false, "Not on players / vehicles" end
    if MAX_OBJECTS > 0 and objCount >= MAX_OBJECTS then p.ok, p.reason = false, "Object limit" end

    if it.kind == "gun" and mountPlan(it, tr, eyeYaw, p) then return p end

    if it.place == "surface" then
        p.ang = surfaceAngle(n, eyeYaw + 180 + rot)
    elseif it.place == "upright" then
        local te, tit = stackTarget(tr, it)
        if te then
            local ta = te:getAngles()
            p.pos = te:localToWorld(V(0, 0, tit.stack))
            p.ang = Angle(0, ta.y + rot, 0)
            p.normal = te:getUp()
            p.stack = true
            return p
        end
        if n.z < 0.6 then p.ok, p.reason = false, "Needs flat ground" end
        local pos = tr.HitPos
        local face = eyeYaw + (it.faceAway and 0 or 180)
        if gridOn and it.grid then
            -- snap position to the grid of the nearest object, yaw: 90 deg steps towards you + R (45)
            local fo, fy, aligned = propFrame(pos)
            local G = PROP_GRIDS[gridIdx]
            local fa = Angle(0, fy, 0)
            local lp = worldToLocal(pos, Angle(), fo, fa)
            local wp = localToWorld(V(snapGrid(lp.x, G), snapGrid(lp.y, G), 0), Angle(), fo, fa)
            local g = groundAt(V(wp.x, wp.y, pos.z))
            pos = g or V(wp.x, wp.y, pos.z)
            p.ang = Angle(0, fy + snapAngle(face - fy, 90) + rot, 0)
            p.grid = { o = fo, yaw = fy, g = G, aligned = aligned }
        else
            p.ang = Angle(0, face + rot, 0)
        end
        p.pos = pos
    elseif it.place == "trench" then
        -- aiming at one of your trenches: take its grid (so you can continue / change old trenches)
        local hitE, hit = regOf(tr.Entity)
        if hitE and hit.place == "trench" then
            local ey = hitE:getAngles().y
            local ep = cellsOf(hitE:getPos(), hitE:getAngles(), hit.size)[1]      -- a cell centre
            local aligned = false
            if trenchOrigin then
                local lp = worldToLocal(ep, Angle(), trenchOrigin, Angle(0, trenchYaw, 0))
                aligned = math.abs(lp.x - snapGrid(lp.x, TRENCH_CELL)) < 4 and math.abs(lp.y - snapGrid(lp.y, TRENCH_CELL)) < 4
                    and math.abs(math.angleDifference(ey, snapAngle(ey - trenchYaw, 90) + trenchYaw)) < 2
            end
            if not aligned then
                trenchOrigin, trenchYaw = ep, ey
                say("Trench grid taken from the trench")
            end
            n = V(0, 0, 1)
        end
        if n.z < 0.7 then p.ok, p.reason = false, "Needs flat ground" end
        if not trenchOrigin then
            p.ang = Angle(0, trenchYaw, 0)
        else
            local ga = Angle(0, trenchYaw, 0)
            local lp = worldToLocal(tr.HitPos, Angle(), trenchOrigin, ga)
            local h2 = (it.size == 2) and TRENCH_CELL / 2 or 0             -- 2x2: centre on a grid corner
            local cx, cy = snapGrid(lp.x - h2, TRENCH_CELL) + h2, snapGrid(lp.y - h2, TRENCH_CELL) + h2
            local wp = localToWorld(V(cx, cy, 0), Angle(), trenchOrigin, ga)
            local g = groundAt(V(wp.x, wp.y, tr.HitPos.z))
            p.pos = g or V(wp.x, wp.y, tr.HitPos.z)
            p.ang = Angle(0, trenchYaw + trenchRot * 90, 0)
            p.cell = { cx, cy }
            local ov = trenchOverlaps(p.pos, p.ang, it.size)
            local old, oit = ov[1] and ov[1].e, ov[1] and ov[1].it
            if #ov == 1 and (oit.size or 1) == (it.size or 1)
                and (old:getPos().x - p.pos.x) ^ 2 + (old:getPos().y - p.pos.y) ^ 2 < 900 then
                -- REPLACE the piece in this cell (keeps the exact position)
                p.pos = old:getPos()
                p.replace = old:entIndex()
                if oit == it and math.abs(math.angleDifference(old:getAngles().y, p.ang.y)) < 2 then
                    p.ok, p.reason = false, "Same piece - R to turn / pick another"
                end
            elseif #ov > 0 then
                p.ok, p.reason = false, "Cells taken (remove the pieces first)"
            end
        end
    end
    return p
end

local function sendPlace(it, p)
    net.start("it_cmd")
    net.writeUInt(1, 4)
    net.writeUInt(it.idx, 8)
    net.writeVector(p.pos)
    net.writeAngle(p.ang or Angle())
    net.writeVector(p.normal or V(0, 0, 1))
    net.writeUInt(p.lvl or 0, 4)
    net.writeUInt(p.replace or 0, 16)
    net.send()
end
local function sendSimple(act)
    net.start("it_cmd")
    net.writeUInt(act, 4)
    net.send()
end

local function okColor(ok, a) return ok and Color(80, 255, 80, a) or Color(255, 60, 60, a) end
local function hideGhosts()

    if ghostGun then ghostGun:setNoDraw(true) end
    if ghostWire then ghostWire:setNoDraw(true) end

hook.add("think", "it_input", function()
    local has = holdsTool(me)
    if not has then
        camViewEnt, camIdx = nil, 0
        if menuOpen then setMenu(false) end
        hideGhosts()
        plan = nil
        return
    end
    local lmb, lmbDown = pressedOnce(MOUSE.MOUSE1)
    local rmb = pressedOnce(MOUSE.MOUSE2)
    local rotP = pressedOnce(KEY_ROTATE)
    local gridP = pressedOnce(KEY_GRID)
    local detP = pressedOnce(KEY_DETONATE)
    local camP = pressedOnce(KEY_CAM)
    if waitRelease then
        if not lmbDown then waitRelease = false end
        lmb = false
    end
    if menuOpen or timer.realtime() < selectorUntil then
        hideGhosts()
        return
    end

    local shift = input.isShiftDown()
    if rotP then
        if selected.place == "trench" then
            if trenchOrigin then trenchRot = (trenchRot + (shift and -1 or 1)) % 4
            else trenchYaw = trenchYaw + (shift and -ROT_FINE or ROT_FINE) end
        else
            rot = rot + (shift and -ROT_STEP or ROT_STEP)
        end
    end
    if gridP then
        if shift and selected.place == "trench" then trenchOrigin = nil trenchRot = 0 say("Trench grid reset")
        elseif shift then
            gridIdx = gridIdx % #PROP_GRIDS + 1
            gridOn = true
            say("Grid " .. PROP_GRIDS[gridIdx])
        else gridOn = not gridOn say(gridOn and "Grid ON" or "Grid OFF") end
    end
    if detP then sendSimple(4) end
    if camP then                                     -- periscope cameras: next / off
        local list = {}
        for idx, r in pairs(reg) do
            if r.it and r.it.logic == "cam" and isValid(entity(idx)) then list[#list + 1] = idx end
        end
        table.sort(list)
        if #list == 0 then say("No cameras placed") camIdx = 0
        else camIdx = (camIdx + 1) % (#list + 1) end
        camViewEnt = camIdx > 0 and entity(list[camIdx]) or nil
        camCount = #list
    end
    if rmb then sendSimple(3) end

    plan = computePlan(selected)
    local showIt = plan.show or selected
    if showIt.kind ~= "cover" then setGhostItem(showIt, plan.lvl) end

    -- ghost visuals
    if plan.wire then
        if ghost then ghost:setNoDraw(true) end
        if ghostGun then ghostGun:setNoDraw(true) end
        if ghostWire and plan.len and plan.len > 1 then
            ghostWire:setNoDraw(false)
            ghostWire:setPos((plan.a + plan.b) / 2)
            ghostWire:setAngles(upFrame(plan.b - plan.a))
            ghostWire:setSize(V(0.6, 0.6, plan.len))
            ghostWire:setColor(okColor(plan.ok, 200))
        end
    else
        if ghostWire then ghostWire:setNoDraw(true) end
        if ghost and plan.pos and ghostItem == showIt then
            ghost:setNoDraw(false)
            if selected.kind == "tripwire" and ghostFix then
                local pp, pa = nadeTransform(ghostFix, plan.pos + plan.normal * (ghostFix.half + 0.3), upFrame(plan.normal))
                ghost:setPos(pp) ghost:setAngles(pa)
            else
                ghost:setPos(plan.pos)
                ghost:setAngles(plan.ang)
            end
            ghost:setColor(okColor(plan.ok, 130))
            if ghostGun then
                if selected.kind == "gun" then
                    local gp, ga = localToWorld(gunPivot(selected.gun, plan.lvl), Angle(), plan.pos, plan.ang)
                    ghostGun:setNoDraw(false)
                    ghostGun:setPos(gp) ghostGun:setAngles(ga)
                    ghostGun:setColor(okColor(plan.ok, 130))
                else
                    ghostGun:setNoDraw(true)
                end
            end
        else
            if ghost then ghost:setNoDraw(true) end
            if ghostGun then ghostGun:setNoDraw(true) end
        end
    end

    if lmb then
        local nowR = timer.realtime()
        if nowR - lastPlaceCl < PLACE_COOLDOWN then return end
        if not plan.ok then say(plan.reason or "Not allowed") return end
        lastPlaceCl = nowR
        if plan.wire then
            net.start("it_cmd")
            net.writeUInt(2, 4)
            net.writeVector(plan.b)
            net.writeVector(plan.normal)
            net.send()
        else
            if selected.place == "trench" and not trenchOrigin then
                trenchOrigin = plan.pos
                trenchRot = 0
            end
            sendPlace(plan.show or selected, plan)
        end
    end
end)

            tri2({ x = sp.x, y = sp.y - 8 }, { x = sp.x - 7, y = sp.y }, { x = sp.x, y = sp.y + 8 })
            render.setRGBA(255, 210, 200, 230)
            render.drawSimpleText(sp.x, sp.y - 16, string.format("%.0fm", p:getDistance(eye) * 0.01905), TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        end
    end
end)

-- hide the bug viewmodel
local vmHidden = nil
hook.add("think", "it_vm", function()
    local want = HIDE_BUGBAIT and holdsTool(me)
    if want == vmHidden then return end
    vmHidden = want
    pcall(function()
        local vm = me:getViewModel()
        if isValid(vm) then vm:setNoDraw(want) end
    end)
end)
